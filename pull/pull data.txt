// In your Laravel controller
public function callbackFromPython($optimizedData) {
    $spoResult = json_decode($optimizedData, true);
    
    // Store the 20 optimized slips
    foreach ($spoResult['final_slips'] as $slip) {
        OptimizedSlip::create([
            'slip_id' => $slip['slip_id'],
            'legs' => json_encode($slip['legs']),
            'total_odds' => $slip['total_odds'],
            'stake' => $slip['stake'],
            'risk_category' => $slip['risk_category'],
            'coverage_role' => $slip['coverage_role'],
            'metadata' => json_encode([
                'true_risk_score' => $slip['true_risk_score'],
                'diversity_score' => $slip['diversity_score']
            ])
        ]);
    }
    
    return response()->json([
        'status' => 'success',
        'metrics' => $spoResult['metrics']
    ]);
}
